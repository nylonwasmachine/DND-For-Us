Je bestaande profiles-tabel blijft gewoon werken. -- ========================================= -- DND FOR US - PARTY SYSTEM -- ========================================= create table public.parties ( id uuid primary key default gen_random_uuid(), name text not null, join_code text unique not null, dm_id uuid not null references public.profiles(id) on delete cascade, created_at timestamp with time zone default now() ); create table public.party_members ( party_id uuid not null references public.parties(id) on delete cascade, user_id uuid not null references public.profiles(id) on delete cascade, joined_at timestamp with time zone default now(), primary key (party_id, user_id) ); -- ========================================= -- ROW LEVEL SECURITY -- ========================================= alter table public.parties enable row level security; alter table public.party_members enable row level security; -- ========================================= -- PARTIES -- ========================================= -- Iedereen die ingelogd is mag parties bekijken create policy "Authenticated users can view parties" on public.parties for select to authenticated using (true); -- Alleen DM's mogen een party maken create policy "DMs can create parties" on public.parties for insert to authenticated with check ( dm_id = auth.uid() and exists ( select 1 from public.profiles where profiles.id = auth.uid() and profiles.role = 'DM' ) ); -- Alleen de eigenaar/DM mag zijn party aanpassen create policy "DMs can update own parties" on public.parties for update to authenticated using (dm_id = auth.uid()) with check (dm_id = auth.uid()); -- Alleen de eigenaar/DM mag zijn party verwijderen create policy "DMs can delete own parties" on public.parties for delete to authenticated using (dm_id = auth.uid()); -- ========================================= -- PARTY MEMBERS -- ========================================= -- Ingelogde gebruikers mogen leden bekijken create policy "Authenticated users can view party members" on public.party_members for select to authenticated using (true); -- Gebruikers mogen zichzelf aan een party toevoegen create policy "Users can join parties" on public.party_members for insert to authenticated with check ( user_id = auth.uid() ); -- Gebruikers mogen zichzelf uit een party verwijderen create policy "Users can leave parties" on public.party_members for delete to authenticated using ( user_id = auth.uid() ); -- ========================================= -- INDEXEN -- ========================================= create index parties_join_code_idx on public.parties(join_code); create index party_members_user_id_idx on public.party_members(user_id); create index party_members_party_id_idx on public.party_members(party_id); Daarna Klik op Run.
